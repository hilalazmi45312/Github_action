/*
|--------------------------------------------------------------------------
| 2.  Output the button
|--------------------------------------------------------------------------
*/
add_action('woocommerce_single_product_summary', function () {
    if (get_option('walc_enable_button') !== 'yes') {
        return;
    }

    global $product;

    if (!$product instanceof WC_Product) {
        return;
    }

    $brand = get_product_brand($product);
    $product_name = $product->get_name();
    $msg_template = "Hi, I'm interested in {$brand} ({$product_name}). Please provide more info";

    $wa_url = 'https://wa.me/' . WALC_PHONE_NUMBER . '?text=' . rawurlencode($msg_template);
    $wa_icon = '<img class="walc-wa-icon" src="' .
        esc_url(plugins_url('assets/uploads/whatsapp-icon.webp', dirname(__FILE__, 2))) .
        '" alt="WhatsApp" />';

    echo '<p><a class="walc-live-chat" href="' . esc_url($wa_url) . '" target="_blank" rel="noopener">' .
        $wa_icon . ' WhatsApp&nbsp;Us</a></p>';
}, 35);


/**
 * Get product brand name from:
 *  1. Product attribute (e.g. pa_brand)
 *  2. Custom field (_brand)
 */
function get_product_brand(WC_Product $product)
{
    // Try attribute
    $attributes = $product->get_attributes();
    foreach (array_keys($attributes) as $attr_key) {
        if (strpos($attr_key, 'brand') !== false) {
            $terms = wp_get_post_terms($product->get_id(), $attr_key, ['fields' => 'names']);
            if (!empty($terms)) {
                return $terms[0];
            }
        }
    }

    // Fallback: check custom field
    $meta = $product->get_meta('_brand', true);
    if (!empty($meta)) {
        return $meta;
    }

    // Final fallback
    return '';
}
